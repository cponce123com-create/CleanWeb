(function() {
  async function sleep(ms) {
    return new Promise(r => setTimeout(r, ms));
  }
  async function run() {
    const photos = document.querySelectorAll(
      'div[data-latest-bg], div[jsmodel], c-wiz div[role="checkbox"]'
    );
    if (!photos.length) {
      alert('No se encontraron fotos. Asegúrate de estar en photos.google.com');
      return;
    }
    let count = 0;
    photos[0].dispatchEvent(new MouseEvent('mouseover', {bubbles: true}));
    await sleep(300);
    photos[0].click();
    await sleep(300);
    photos.forEach(p => { p.click(); count++; });
    await sleep(500);
    const moreBtn = 
      document.querySelector('button[aria-label="More options"]') ||
      document.querySelector('button[aria-label="Más opciones"]');
    if (moreBtn) moreBtn.click();
    await sleep(500);
    const deleteBtn = Array.from(document.querySelectorAll('li[role="menuitem"]'))
      .find(el => el.textContent.includes('Delete') ||
                  el.textContent.includes('Eliminar') ||
                  el.textContent.includes('papelera'));
    if (deleteBtn) {
      deleteBtn.click();
      await sleep(500);
      const confirmBtn = 
        document.querySelector('button[data-mdc-dialog-action="ok"]') ||
        Array.from(document.querySelectorAll('button'))
          .find(b => b.textContent.includes('Move to trash') ||
                     b.textContent.includes('Mover a la papelera'));
      if (confirmBtn) {
        confirmBtn.click();
        await sleep(1000);
        alert(`✓ ${count} fotos enviadas a la papelera.\nRecarga con F5 y ejecuta el script de nuevo.`);
      } else {
        alert('No se encontró el botón de confirmar. Inténtalo de nuevo.');
      }
    } else {
      alert('No se encontró el botón de eliminar. Inténtalo de nuevo.');
    }
  }
  run();
})();

(function() {
  const items = document.querySelectorAll('ytd-playlist-video-renderer');
  if (!items.length) {
    alert('No se encontraron videos. Asegúrate de estar en una página de playlist de YouTube y haber cargado todos los videos.');
    return;
  }
  const rows = [['#', 'Título', 'Canal', 'URL']];
  items.forEach((item, i) => {
    const titleEl = item.querySelector('#video-title');
    const channelEl = item.querySelector('#channel-name a, .ytd-channel-name a');
    const title = titleEl ? titleEl.textContent.trim() : 'Sin título';
    const channel = channelEl ? channelEl.textContent.trim() : 'Desconocido';
    const url = titleEl ? 'https://www.youtube.com' + titleEl.getAttribute('href') : '';
    rows.push([i + 1, title, channel, url]);
  });
  const csv = rows.map(r => r.map(c => `"${String(c).replace(/"/g,'""')}"`).join(',')).join('\n');
  const blob = new Blob(['\uFEFF' + csv], {type: 'text/csv;charset=utf-8;'});
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'youtube-playlist.csv';
  a.click();
  alert(`✓ ${items.length} videos exportados a youtube-playlist.csv`);
})();
- Tips:

  • El CSV se abre con Excel, Google Sheets o Numbers

(async function() {
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const random = (min, max) => Math.floor(Math.random() * (max - min + 1)) + min;

  function getCookie(name) {
    const v = document.cookie.match('(^|;) ?' + name + '=([^;]*)(;|$)');
    return v ? v[2] : null;
  }

  const csrftoken = getCookie('csrftoken');
  const ds_user_id = getCookie('ds_user_id');

  if (!csrftoken || !ds_user_id) {
    alert('No se pudo obtener la sesión. Asegúrate de estar en instagram.com y haber iniciado sesión.');
    return;
  }

  let after = null;
  let totalUnfollowed = 0;
  let doNext = true;

  while (doNext) {
    const url = after
      ? `https://www.instagram.com/graphql/query/?query_hash=3dec7e2c57367ef3da3d987d89f9dbc8&variables={"id":"${ds_user_id}","first":"24","after":"${after}"}`
      : `https://www.instagram.com/graphql/query/?query_hash=3dec7e2c57367ef3da3d987d89f9dbc8&variables={"id":"${ds_user_id}","first":"24"}`;

    let data;
    try {
      const res = await fetch(url);
      data = await res.json();
    } catch(e) {
      console.log('Error fetching:', e);
      break;
    }

    const following = data?.data?.user?.edge_follow;
    if (!following) { alert('Error al obtener lista. Intenta de nuevo.'); break; }

    const users = following.edges;
    doNext = following.page_info.has_next_page;
    after = following.page_info.end_cursor;

    for (const { node } of users) {
      if (!node.follows_viewer) {
        try {
          await fetch(`https://www.instagram.com/web/friendships/${node.id}/unfollow/`, {
            method: 'POST',
            headers: {
              'content-type': 'application/x-www-form-urlencoded',
              'x-csrftoken': csrftoken
            },
            credentials: 'include'
          });
          totalUnfollowed++;
          console.log(`✓ Unfollowed: ${node.username} (${totalUnfollowed})`);
        } catch(e) {
          console.log(`✗ Error unfollowing ${node.username}`);
        }
        await sleep(random(3000, 6000));
      }
    }
    if (doNext) await sleep(random(5000, 10000));
  }
  alert(`✓ Proceso completado. Se dejaron de seguir ${totalUnfollowed} usuarios.`);
})();

(async function() {
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const random = (min, max) => Math.floor(Math.random() * (max - min + 1)) + min;
  let count = 0;

  async function unlikeVisible() {
    const likedButtons = Array.from(document.querySelectorAll('button'))
      .filter(b => b.getAttribute('aria-label') === 'Unlike' || 
                   b.getAttribute('aria-label') === 'Ya no me gusta');
    
    for (const btn of likedButtons) {
      btn.click();
      count++;
      console.log(`✓ Unlike #${count}`);
      await sleep(random(1500, 3000));
    }
    return likedButtons.length;
  }

  async function run() {
    let found = await unlikeVisible();
    if (found === 0) {
      alert('No se encontraron likes visibles. Asegúrate de estar en la página de Actividad → Me gusta de Instagram.');
      return;
    }
    window.scrollTo(0, document.body.scrollHeight);
    await sleep(2000);
    const more = await unlikeVisible();
    alert(`✓ ${count} likes eliminados en esta sesión. Recarga y ejecuta de nuevo para continuar.`);
  }

  run();
})();

      '[data-testid$="-unfollow"], button[aria-label*="Following"]'

    for (const btn of buttons) {

        return false;
      }
      btn.click();

    window.scrollTo(0, document.body.scrollHeight);
    await sleep(2000);
  }
  alert(`✓ Proceso completado. Se dejaron de seguir ${count} usuarios.`);

async function generateCodeVerifier(length) {
  const array = new Uint8Array(length);
  window.crypto.getRandomValues(array);
  return btoa(String.fromCharCode(...array))
    .replace(/\+/g, '-').replace(/\//g, '_').replace(/=/g, '');
}
async function generateCodeChallenge(verifier) {
  const data = new TextEncoder().encode(verifier);
  const digest = await window.crypto.subtle.digest('SHA-256', data);
  return btoa(String.fromCharCode(...new Uint8Array(digest)))
    .replace(/\+/g, '-').replace(/\//g, '_').replace(/=/g, '');
}
